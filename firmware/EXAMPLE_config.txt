diceId=EXAMPLE

# Below are settings for Advanced users. Default values are ok to use.

# ==========================================
# DISPLAY COLORS
# ==========================================
# 16-bit RGB565 format (0-65535)
# Common colors:
# - Red: 63488 (0xF800)
# - Yellow: 65504 (0xFFE0)
# - Green: 2016 (0x07E0)
# - Cyan: 2047 (0x07FF)
# - Blue: 31 (0x001F)
# - Magenta: 63519 (0xF81F)
# - White: 65535 (0xFFFF)
# - Black: 0 (0x0000)
x_background=0 # Black
y_background=0 # Black
z_background=0 # Black

# ==========================================
# ENTANGLEMENT ANIMATION
# ==========================================
# The duration of the animation is milliseconds.
animationDuration=2000

# The number of spins done during the animation.
animationSpins=2

# ==========================================
# OBSERVED TIME
# ==========================================
# The time after a measurement has been made during which the dice does not do anything, but display the measurement.
# Time in milliseconds.
observedTime=2000

# ==========================================
# RSSI SETTINGS
# ==========================================
# The RSSI close threshold determines when a dice is willing to entangle to another.
# More negative = weaker signal allowed.
# rssiClose=-35 (pre Gen 4.0)
rssiClose=-10

# The RSSI far threshold indicates how far two dice need to be before they can interact again.
# Set to 0 to disable.
rssiFar=0

# ==========================================
# OPERATIONAL PARAMETERS
# ==========================================
# Deep sleep timeout (milliseconds)
# Time of inactivity before entering deep sleep
# 300000 = 5 minutes, 600000 = 10 minutes
deepSleepTimeout=300000

# ==========================================
# CHECKSUM
# ==========================================
# Auto-calculated when saving from ESP32
# Set to 0 to disable validation
checksum=0
